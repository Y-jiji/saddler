# Global registry for whitelist.sh: the paths visible inside the sandbox
# besides the project, library and PATH directories. One `ro PATH`, `rw PATH`,
# `private PATH` or `guard PATH` per line, the path being the rest of the line,
# ~ expanded, missing paths skipped. `guard PATH` makes every .claude below PATH,
# outside a folder holding its own .git, read-only.
# A socket in a `ro` or `rw` line is refused, because read-only cannot restrict
# it. `dbus-system NAME RULE` lets the system bus, through xdg-dbus-proxy, take
# calls matching RULE, `[METHOD][@PATH]`, on NAME, and nothing else.
# `private PATH` gives a folder a private copy: commands read the real one and
# write to a throwaway upper, gone when the sandbox ends.
ro /usr
ro /etc
ro /opt
ro /sys
ro /var
ro ~/.gitconfig
ro ~/.config/git
ro ~/.npmrc
ro ~/.claude/shell-snapshots
rw /tmp
rw ~/.cache
rw ~/.cargo/registry
rw ~/.cargo/git
rw ~/.npm
rw ~/.bun/install/cache
rw ~/.gradle
rw ~/.m2
rw ~/.elan
rw ~/.config/gcloud
private ~/.local/state/nvim
dbus-system org.bluez org.freedesktop.DBus.Introspectable.Introspect@/
dbus-system org.bluez org.freedesktop.DBus.ObjectManager.GetManagedObjects@/
dbus-system org.freedesktop.NetworkManager org.freedesktop.DBus.Introspectable.Introspect@/org/freedesktop/NetworkManager/*
dbus-system org.freedesktop.NetworkManager org.freedesktop.NetworkManager.GetDevices@/org/freedesktop/NetworkManager
dbus-system org.freedesktop.NetworkManager org.freedesktop.DBus.Properties.Get@/org/freedesktop/NetworkManager/*
